# ------------------------------------------------------------
# AppArmor‑Profil für Conduit (Matrix‑Server)
# ------------------------------------------------------------
#include <tunables/global>

profile portainer flags=(attach_disconnected,mediate_deleted) {
  # ------------------------------------------------------------------
  # Basis‑Abstraktionen
  # ------------------------------------------------------------------
  # Standard‑Linux‑Abstraktionen (Dateisystem, Netzwerk, etc.)
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/user-tmp>
  #include <abstractions/python>   # falls Conduit Python‑Bindings nutzt

  # ------------------------------------------------------------------
  # Capabilities
  # ------------------------------------------------------------------
  capability,
  capability setgid,
  capability setuid,
  capability sys_admin,
  capability net_bind_service,
  capability dac_read_search,
  capability dac_override,

  # ------------------------------------------------------------------
  # Netzwerk‑Zugriff
  # ------------------------------------------------------------------
  network udp,
  network tcp,
  network dgram,
  network stream,
  network inet raw,
  network inet6 raw,
  network netlink raw,
  network unix dgram,

  # ------------------------------------------------------------------
  # Binärdatei (Ausführungsrechte)
  # ------------------------------------------------------------------
  /usr/local/bin/matrix-conduit ix,   # die eigentliche Binary
  /usr/local/bin/conduit      ix,     # falls ein Symlink/Wrapper existiert

  # ------------------------------------------------------------------
  # Datenverzeichnis
  # ------------------------------------------------------------------
  /data/conduit/**           rw,      # komplette Read/Write‑Erlaubnis

  # ------------------------------------------------------------------
  # S6‑Overlay (wie in deinem Original‑Snippet)
  # ------------------------------------------------------------------
  /init                      ix,
  /bin/**                    ix,
  /usr/bin/**                ix,
  /etc/s6/**                 rix,
  /run/s6/**                 rix,
  /etc/services.d/**         rwix,
  /etc/cont-init.d/**        rwix,
  /etc/cont-finish.d/**      rwix,
  /run/**                    rwk,

  # ------------------------------------------------------------------
  # Bashio (falls du Home‑Assistant‑Helper nutzt)
  # ------------------------------------------------------------------
  /usr/lib/bashio/**         ix,
  /tmp/**                    rw,

  # ------------------------------------------------------------------
  # .NET‑Runtime (falls du Conduit über dotnet startest)
  # ------------------------------------------------------------------
  /usr/bin/dotnet            ix,

  # ------------------------------------------------------------------
  # ptrace‑Ausnahme (Docker‑Tools)
  # ------------------------------------------------------------------
  ptrace (trace,read) peer=docker-default,

  # ------------------------------------------------------------------
  # Signal‑Ausnahme (Docker‑Daemon)
  # ------------------------------------------------------------------
  signal (receive) set=(kill,term) peer=/usr/bin/docker,

  # ------------------------------------------------------------------
  # Optional: Home‑Verzeichnis des Service‑Users (falls genutzt)
  # ------------------------------------------------------------------
  owner @{HOME}/**           rw,

  # ------------------------------------------------------------------
  # Default‑Verhalten – alles andere verweigern
  # ------------------------------------------------------------------
  deny /**                   w,
  deny /**                   m,
}
